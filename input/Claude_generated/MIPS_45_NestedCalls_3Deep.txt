int inc(int x) { return x + 1; }
int dbl(int x) { return x * 2; }
int sqr(int x) { return x * x; }
int myNeg(int x) { return 0 - x; }

void main()
{
    PrintInt(inc(dbl(sqr(3))));
    PrintInt(dbl(inc(sqr(4))));
    PrintInt(sqr(dbl(inc(2))));
    PrintInt(myNeg(dbl(inc(5))));
    PrintInt(inc(inc(inc(inc(0)))));
    int a := 3;
    int b := sqr(dbl(a));
    int c := dbl(sqr(a));
    PrintInt(b);
    PrintInt(c);
    PrintInt(b - c);
}

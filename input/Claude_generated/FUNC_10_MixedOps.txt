class Calculator
{
    int lastResult := 0;
    int add(int a, int b)
    {
        lastResult := a + b;
        return lastResult;
    }
    int multiply(int a, int b)
    {
        lastResult := a * b;
        return lastResult;
    }
    int chain(int a, int b, int c)
    {
        int temp := add(a, b);
        return multiply(temp, c);
    }
}

string dayType(int day)
{
    if (day = 1)
    {
        return "Mon";
    }
    else
    {
        if (day = 2)
        {
            return "Tue";
        }
        else
        {
            if (day = 3)
            {
                return "Wed";
            }
            else
            {
                if (day = 4)
                {
                    return "Thu";
                }
                else
                {
                    if (day = 5)
                    {
                        return "Fri";
                    }
                    else
                    {
                        return "Weekend";
                    }
                }
            }
        }
    }
}

void main()
{
    PrintString(dayType(1));
    PrintString(dayType(3));
    PrintString(dayType(5));
    PrintString(dayType(6));
    PrintString(dayType(7));
}

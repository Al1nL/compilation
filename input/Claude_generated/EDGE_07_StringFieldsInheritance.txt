class Animal
{
    string name := "animal";
    string sound := "none";
    string describe() { return name + sound; }
    string getName() { return name; }
}
class Dog extends Animal
{
    string trick := "sit";
    string describe() { return name + sound + trick; }
}
class Cat extends Animal
{
    string describe() { return name + name; }
}

string callDescribe(Animal a) { return a.describe(); }

void main()
{
    Animal a := new Animal;
    Dog d := new Dog;
    Cat c := new Cat;
    d.name := "Rex";
    d.sound := "woof";
    c.name := "Whiskers";
    c.sound := "meow";
    PrintString(a.describe());
    PrintString(d.describe());
    PrintString(c.describe());
    PrintString(callDescribe(a));
    PrintString(callDescribe(d));
    PrintString(callDescribe(c));
    int eq1 := d.getName() = "Rex";
    int eq2 := c.getName() = "Whiskers";
    PrintInt(eq1);
    PrintInt(eq2);
}

class Box
{
    int val := 0;
}

void main()
{
    Box b1 := new Box;
    Box b2 := new Box;
    Box b3 := new Box;
    b1.val := 111;
    b2.val := 222;
    b3.val := 333;
    PrintInt(b1.val);
    PrintInt(b2.val);
    PrintInt(b3.val);
    b1.val := 999;
    PrintInt(b1.val);
    PrintInt(b2.val);
    PrintInt(b3.val);
    Box b4 := b2;
    b4.val := 777;
    PrintInt(b2.val);
    PrintInt(b4.val);
    int same := b2 = b4;
    int diff := b1 = b2;
    PrintInt(same);
    PrintInt(diff);
}

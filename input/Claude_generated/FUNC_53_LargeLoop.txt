void main()
{
    int i := 0;
    int sum := 0;
    while (i < 1000)
    {
        sum := sum + 1;
        i := i + 1;
    }
    PrintInt(sum);
    int product := 1;
    int j := 1;
    while (j < 16)
    {
        product := product * j;
        if (product > 30000)
        {
            product := 30000;
        }
        j := j + 1;
    }
    PrintInt(product);
    int count := 0;
    int k := 1;
    while (k < 100)
    {
        if ((k / 3) * 3 = k)
        {
            count := count + 1;
        }
        k := k + 1;
    }
    PrintInt(count);
}

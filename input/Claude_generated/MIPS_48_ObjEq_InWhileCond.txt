class Node
{
    int val := 0;
    Node next := nil;
}
void main()
{
    Node sentinel := new Node;
    Node curr := new Node;
    Node n2 := new Node;
    Node n3 := new Node;
    curr.val := 1; curr.next := n2;
    n2.val := 2;   n2.next := n3;
    n3.val := 3;   n3.next := sentinel;
    int sum := 0;
    while ((curr = sentinel) = 0)
    {
        sum := sum + curr.val;
        curr := curr.next;
    }
    PrintInt(sum);
    PrintInt(curr = sentinel);
    Node a := new Node;
    Node b := a;
    while (a = b)
    {
        a.val := a.val + 1;
        if (a.val > 3)
        {
            b := new Node;
        }
    }
    PrintInt(a.val);
}

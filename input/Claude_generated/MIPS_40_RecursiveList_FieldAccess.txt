class Node
{
    int val := 0;
    Node next := nil;
}
int listSum(Node n)
{
    if (n = nil) { return 0; }
    return n.val + listSum(n.next);
}
int listLen(Node n)
{
    if (n = nil) { return 0; }
    return 1 + listLen(n.next);
}
void main()
{
    Node n1 := new Node;
    Node n2 := new Node;
    Node n3 := new Node;
    Node n4 := new Node;
    n1.val := 10; n2.val := 20; n3.val := 30; n4.val := 40;
    n1.next := n2;
    n2.next := n3;
    n3.next := n4;
    PrintInt(listLen(n1));
    PrintInt(listSum(n1));
    PrintInt(listLen(n2));
    PrintInt(listSum(n2));
    PrintInt(listLen(nil));
    PrintInt(listSum(nil));
}

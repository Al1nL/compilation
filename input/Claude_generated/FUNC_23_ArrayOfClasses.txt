class Point
{
    int x := 0;
    int y := 0;
    int distSq()
    {
        return x * x + y * y;
    }
}

array PointArray = Point[];

void main()
{
    PointArray pts := new Point[4];
    pts[0] := new Point;
    pts[1] := new Point;
    pts[2] := new Point;
    pts[3] := new Point;
    pts[0].x := 3;
    pts[0].y := 4;
    pts[1].x := 0;
    pts[1].y := 5;
    pts[2].x := 5;
    pts[2].y := 0;
    pts[3].x := 1;
    pts[3].y := 1;
    int i := 0;
    while (i < 4)
    {
        PrintInt(pts[i].distSq());
        i := i + 1;
    }
    PrintInt(pts[0].x + pts[1].x + pts[2].x + pts[3].x);
}

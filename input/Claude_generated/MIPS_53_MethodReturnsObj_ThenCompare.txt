class Node
{
    int val := 0;
    Node next := nil;
    Node getNext() { return next; }
    int getVal() { return val; }
}

void main()
{
    Node a := new Node;
    Node b := new Node;
    Node c := new Node;
    a.val := 1; b.val := 2; c.val := 3;
    a.next := b;
    b.next := c;
    Node aNext := a.getNext();
    Node bNext := b.getNext();
    Node aNextNext := aNext.getNext();
    PrintInt(aNext = b);
    PrintInt(aNext = c);
    PrintInt(bNext = c);
    PrintInt(aNext.getVal());
    PrintInt(bNext.getVal());
    PrintInt(aNextNext.getVal());
    PrintInt(aNextNext = c);
    int nilCheck := aNextNext.getNext() = nil;
    PrintInt(nilCheck);
}

int factorial(int n)
{
    if (n < 2)
    {
        return 1;
    }
    return n * factorial(n-1);
}
int fibonacci(int n)
{
    if (n < 2)
    {
        return n;
    }
    return fibonacci(n-1) + fibonacci(n-2);
}
int ackermann(int m, int n)
{
    if (m = 0)
    {
        return n + 1;
    }
    if (n = 0)
    {
        return ackermann(m-1, 1);
    }
    return ackermann(m-1, ackermann(m, n-1));
}
int isEven(int n)
{
    if (n = 0)
    {
        return 1;
    }
    if (n = 1)
    {
        return 0;
    }
    return isEven(n-2);
}
int isOdd(int n)
{
    if (n = 0)
    {
        return 0;
    }
    if (n = 1)
    {
        return 1;
    }
    return isOdd(n-2);
}
void main()
{
    PrintInt(factorial(5));
    PrintInt(factorial(7));
    PrintInt(fibonacci(8));
    PrintInt(fibonacci(10));
    PrintInt(ackermann(2, 2));
    PrintInt(ackermann(3, 1));
    PrintInt(isEven(0));
    PrintInt(isEven(1));
    PrintInt(isOdd(0));
    PrintInt(isOdd(1));
    PrintInt(isEven(10));
    PrintInt(isOdd(15));
}

class A
{
    int x := 1;
    int getX() { return x; }
    int doubled() { return x * 2; }
}
class B extends A
{
    int doubled() { return x * 3; }
}
class C extends B
{
    int getX() { return x + 100; }
}
class D extends C
{
    int w := 0;
}
void main()
{
    A oa := new A;
    A ob := new B;
    A oc := new C;
    A od := new D;
    oa.x := 5; ob.x := 5; oc.x := 5; od.x := 5;
    PrintInt(oa.getX());
    PrintInt(ob.getX());
    PrintInt(oc.getX());
    PrintInt(od.getX());
    PrintInt(oa.doubled());
    PrintInt(ob.doubled());
    PrintInt(oc.doubled());
    PrintInt(od.doubled());
}

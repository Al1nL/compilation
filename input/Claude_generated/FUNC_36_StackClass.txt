class StackNode
{
    int data := 0;
    StackNode prev := nil;
}

class Stack
{
    StackNode top := nil;
    int size := 0;

    int push(int val)
    {
        StackNode n := new StackNode;
        n.data := val;
        n.prev := top;
        top := n;
        size := size + 1;
        return val;
    }

    int pop()
    {
        int val := top.data;
        top := top.prev;
        size := size - 1;
        return val;
    }

    int peek()
    {
        return top.data;
    }

    int getSize()
    {
        return size;
    }
}

void main()
{
    Stack s := new Stack;
    s.push(10);
    s.push(20);
    s.push(30);
    PrintInt(s.getSize());
    PrintInt(s.peek());
    PrintInt(s.pop());
    PrintInt(s.getSize());
    PrintInt(s.peek());
    s.push(99);
    PrintInt(s.peek());
    PrintInt(s.pop());
    PrintInt(s.pop());
    PrintInt(s.pop());
    PrintInt(s.getSize());
}

array IntArray = int[];
array StringArray = string[];
array IntArrayArray = IntArray[];
int sumArray(IntArray arr, int size)
{
    int sum := 0;
    int i := 0;
    while (i < size)
    {
        sum := sum + arr[i];
        i := i + 1;
    }
    return sum;
}
void fillArray(IntArray arr, int size, int value)
{
    int i := 0;
    while (i < size)
    {
        arr[i] := value + i;
        i := i + 1;
    }
}

    IntArray nums := new int[10];
    fillArray(nums, 10, 100);
    PrintInt(nums[0]);
    PrintInt(nums[5]);
    PrintInt(nums[9]);
    PrintInt(sumArray(nums, 10));
    IntArrayArray matrix := new IntArray[3];
    matrix[0] := new int[5];
    matrix[1] := new int[5];
    matrix[2] := new int[5];
    int i := 0;
    while (i < 3)
    {
        int j := 0;
        while (j < 5)
        {
            matrix[i][j] := i * 10 + j;
            j := j + 1;
        }
        i := i + 1;
    }
    PrintInt(matrix[1][3]);
    PrintInt(matrix[2][4]);
    StringArray words := new string[3];
    words[0] := "Hello";
    words[1] := "World";
    words[2] := words[0] + words[1];
    PrintString(words[2]);
    IntArray arr1 := new int[5];

array IntArray = int[];

void doubleAll(IntArray a, int len)
{
    int i := 0;
    while (i < len)
    {
        a[i] := a[i] * 2;
        i := i + 1;
    }
}

int findMax(IntArray a, int len)
{
    int m := a[0];
    int i := 1;
    while (i < len)
    {
        if (a[i] > m)
        {
            m := a[i];
        }
        i := i + 1;
    }
    return m;
}

void main()
{
    IntArray arr := new int[5];
    arr[0] := 3;
    arr[1] := 1;
    arr[2] := 4;
    arr[3] := 1;
    arr[4] := 5;
    PrintInt(findMax(arr, 5));
    doubleAll(arr, 5);
    PrintInt(arr[0]);
    PrintInt(arr[2]);
    PrintInt(arr[4]);
    PrintInt(findMax(arr, 5));
}

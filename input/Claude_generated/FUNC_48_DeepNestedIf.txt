int classify(int n)
{
    if (n < 0)
    {
        return 0;
    }
    if (n < 10)
    {
        if (n < 5)
        {
            if (n < 2)
            {
                return 1;
            }
            return 2;
        }
        return 3;
    }
    if (n < 100)
    {
        return 4;
    }
    return 5;
}

void main()
{
    PrintInt(classify(0 - 5));
    PrintInt(classify(0));
    PrintInt(classify(1));
    PrintInt(classify(3));
    PrintInt(classify(7));
    PrintInt(classify(50));
    PrintInt(classify(200));
}

array IntArray = int[];

void encodeArray(IntArray arr, int len, int shift)
{
    int i := 0;
    while (i < len)
    {
        arr[i] := arr[i] + shift;
        i := i + 1;
    }
}

int dotProduct(IntArray a, IntArray b, int len)
{
    int result := 0;
    int i := 0;
    while (i < len)
    {
        result := result + a[i] * b[i];
        i := i + 1;
    }
    return result;
}

void main()
{
    IntArray v1 := new int[4];
    IntArray v2 := new int[4];
    v1[0] := 1; v1[1] := 2; v1[2] := 3; v1[3] := 4;
    v2[0] := 5; v2[1] := 6; v2[2] := 7; v2[3] := 8;
    PrintInt(dotProduct(v1, v2, 4));
    encodeArray(v1, 4, 10);
    PrintInt(v1[0]);
    PrintInt(v1[3]);
    PrintInt(dotProduct(v1, v2, 4));
}

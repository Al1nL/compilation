class A { int x := 0; }
class B { int y := 0; }
class C { int z := 0; }
void main()
{
    A a := new A;
    B b := new B;
    C c := new C;
    a.x := 10;
    b.y := 20;
    c.z := 30;
    int sum := a.x + b.y + c.z;
    PrintInt(sum);
    a.x := a.x * 2;
    b.y := b.y + a.x;
    c.z := c.z + b.y;
    PrintInt(a.x);
    PrintInt(b.y);
    PrintInt(c.z);
    int eq1 := a.x = b.y;
    int eq2 := b.y = c.z;
    PrintInt(eq1);
    PrintInt(eq2);
}
